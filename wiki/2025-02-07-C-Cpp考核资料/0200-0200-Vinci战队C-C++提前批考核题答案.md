#### 输入一个字符串\(长度小于100\)，并将这个字符串逆序输出。\(提示:可以使用 cin\.getline\(\) 输入带有空格的字符串\)

![image\.png](https://cdn.sdutvincirobot.top/site-assets/images/wiki/2025/02/07/26d02450e857657c.webp)

```C++
#include <iostream>
using namespace std;

int main()
{
    char str[100] = {0};
    cin.getline(str, 100);

    for (int i = strlen(str); i >= 0 ; i--) {
        cout << str[i];
    }
        return 0;
}
```

#### 定义两个数组 int a\[n\] 和 int b\[m\]，定义一个函数exchange，按顺序交换这两个数组中的元素并打印。

![image\.png](https://cdn.sdutvincirobot.top/site-assets/images/wiki/2025/02/07/6a1dcf2c2c2d8359.webp)

```C++
#include <iostream>
using namespace std;

void exchange(int *a,int *b)；

int main()
{
    //int n;
    //cin >> n;
    int a[] = { 1,2,3,4 }, b[] = { 5,6,7,8 };
    cout << "交换元素之前：" << endl<<" a：";
    for (int i = 0; i < 4; i++) {
        cout << a[i] << " ";
    }
    cout << "   b：";
    for (int i = 0; i < 4; i++) {
        cout << b[i] << " ";
    }
    cout << endl;
    exchange(a, b);
    cout << "交换元素之后：" << endl << " a：";
    for (int i = 0; i < 4; i++) {
        cout << a[i] << " ";
    }
    cout << "  b：";
    for (int i = 0; i < 4; i++) {
        cout << b[i] << " ";
    }
    return 0;
}

    void exchange(int *a,int *b) {
    int t = 0;
    for (int i = 0; i < 4; i++) {
        t = *a;
        *a = *b;
        *b = t;
        a++;
        b++;
    }
}
```
